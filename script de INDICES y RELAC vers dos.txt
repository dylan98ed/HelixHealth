-- ================================================================
-- 1. VERIFICAR QUE TODAS LAS TABLAS EXISTEN

SELECT '=== VERIFICACIÓN DE TABLAS ===' as '';

SELECT name as 'Tabla Existente' 
FROM sqlite_master 
WHERE type='table' 
AND name IN (
    'Especialidades', 'Profesionales', 'Pacientes', 
    'SignosVitales', 'SnomedCT', 'Farmacos', 'Prescripciones'
)
ORDER BY name;

-- ================================================================
-- 3. VERIFICAR COLUMNA 'activo' EN TODAS LAS TABLAS
-- ================================================================

SELECT '=== VERIFICACIÓN DE COLUMNA ACTIVO ===' as '';

SELECT 
    m.name as Tabla,
    CASE 
        WHEN EXISTS (
            SELECT 1 FROM pragma_table_info(m.name) 
            WHERE name = 'activo'
        ) THEN '✅ SÍ'
        ELSE '❌ NO'
    END as 'Tiene activo'
FROM sqlite_master m
WHERE type = 'table' 
AND name IN (
    'Pacientes', 'Profesionales', 'SignosVitales', 
    'Prescripciones', 'Especialidades', 'SnomedCT', 'Farmacos'
)
ORDER BY m.name;


-- ================================================================
-- 4. VERIFICAR CLAVES FORÁNEAS
-- ================================================================

SELECT '=== CLAVES FORÁNEAS ===' as '';

-- Verificar si las FK están habilitadas
SELECT 'Foreign Keys activas:' as 'Estado';
PRAGMA foreign_keys;

-- FK de Profesionales
SELECT '--- FK Profesionales → Especialidades ---' as '';

SELECT '--- FK Prescripciones ---' as '';
PRAGMA foreign_key_list(Prescripciones);


-- ================================================================
-- 5. VERIFICAR ÍNDICES EXISTENTES
-- ================================================================

SELECT '=== ÍNDICES EXISTENTES ===' as '';

-- Todos los índices
SELECT 
    name as 'Índice',
    tbl_name as 'Tabla',
    CASE 
        WHEN sql LIKE '%UNIQUE%' THEN 'UNIQUE'
        ELSE 'Normal'
    END as 'Tipo'
FROM sqlite_master 
WHERE type='index' 
AND name NOT LIKE 'sqlite_%'
ORDER BY tbl_name, name;


-- ================================================================
-- 6. CONTAR REGISTROS POR TABLA (ACTIVOS/INACTIVOS)
-- ================================================================

SELECT '=== CONTEO DE REGISTROS ===' as '';

-- Pacientes
SELECT '--- PACIENTES ---' as '';
SELECT 
    COUNT(*) as Total,
    SUM(CASE WHEN activo = 1 THEN 1 ELSE 0 END) as Activos,
    SUM(CASE WHEN activo = 0 THEN 1 ELSE 0 END) as Inactivos
FROM Pacientes;

-- Profesionales
SELECT '--- PROFESIONALES ---' as '';
SELECT 
    COUNT(*) as Total,
    SUM(CASE WHEN activo = 1 THEN 1 ELSE 0 END) as Activos,
    SUM(CASE WHEN activo = 0 THEN 1 ELSE 0 END) as Inactivos
FROM Profesionales;

-- SignosVitales
SELECT '--- SIGNOS VITALES ---' as '';
SELECT 
    COUNT(*) as Total,
    SUM(CASE WHEN activo = 1 THEN 1 ELSE 0 END) as Activos,
    SUM(CASE WHEN activo = 0 THEN 1 ELSE 0 END) as Inactivos
FROM SignosVitales;

-- Prescripciones
SELECT '--- PRESCRIPCIONES ---' as '';
SELECT 
    COUNT(*) as Total,
    SUM(CASE WHEN activo = 1 THEN 1 ELSE 0 END) as Activas,
    SUM(CASE WHEN activo = 0 THEN 1 ELSE 0 END) as Anuladas
FROM Prescripciones;

-- Especialidades
SELECT '--- ESPECIALIDADES ---' as '';
SELECT 
    COUNT(*) as Total,
    SUM(CASE WHEN activo = 1 THEN 1 ELSE 0 END) as Activas,
    SUM(CASE WHEN activo = 0 THEN 1 ELSE 0 END) as Inactivas
FROM Especialidades;

-- SnomedCT
SELECT '--- SNOMED CT ---' as '';
SELECT 
    COUNT(*) as Total,
    SUM(CASE WHEN activo = 1 THEN 1 ELSE 0 END) as Activos,
    SUM(CASE WHEN activo = 0 THEN 1 ELSE 0 END) as Inactivos
FROM SnomedCT;

-- Farmacos
SELECT '--- FÁRMACOS ---' as '';
SELECT 
    COUNT(*) as Total,
    SUM(CASE WHEN activo = 1 THEN 1 ELSE 0 END) as Activos,
    SUM(CASE WHEN activo = 0 THEN 1 ELSE 0 END) as Inactivos
FROM Farmacos;


-- ================================================================
-- 7. VERIFICAR INTEGRIDAD REFERENCIAL
-- ================================================================

SELECT '=== VERIFICACIÓN DE INTEGRIDAD REFERENCIAL ===' as '';

-- Profesionales sin especialidad válida
SELECT '--- Profesionales con especialidad inválida ---' as '';
SELECT p.id, p.dni, p.nombre, p.apellido, p.especialidad_id
FROM Profesionales p
LEFT JOIN Especialidades e ON p.especialidad_id = e.id
WHERE p.especialidad_id IS NOT NULL AND e.id IS NULL;

-- SignosVitales sin paciente válido
SELECT '--- SignosVitales sin paciente válido ---' as '';
SELECT s.id, s.paciente_id 
FROM SignosVitales s
LEFT JOIN Pacientes p ON s.paciente_id = p.id
WHERE p.id IS NULL;

-- SignosVitales sin médico válido
SELECT '--- SignosVitales sin médico válido ---' as '';
SELECT s.id, s.medico_id 
FROM SignosVitales s
LEFT JOIN Profesionales p ON s.medico_id = p.id
WHERE s.medico_id IS NOT NULL AND p.id IS NULL;

-- Prescripciones sin paciente válido
SELECT '--- Prescripciones sin paciente válido ---' as '';
SELECT pr.id, pr.paciente_id 
FROM Prescripciones pr
LEFT JOIN Pacientes p ON pr.paciente_id = p.id
WHERE p.id IS NULL;

-- Prescripciones sin profesional válido
SELECT '--- Prescripciones sin profesional válido ---' as '';
SELECT pr.id, pr.profesional_id 
FROM Prescripciones pr
LEFT JOIN Profesionales p ON pr.profesional_id = p.id
WHERE p.id IS NULL;

-- Prescripciones sin fármaco válido
SELECT '--- Prescripciones sin fármaco válido ---' as '';
SELECT pr.id, pr.farmaco_id 
FROM Prescripciones pr
LEFT JOIN Farmacos f ON pr.farmaco_id = f.id
WHERE f.id IS NULL;

-- Prescripciones con SNOMED inválido
SELECT '--- Prescripciones con SNOMED inválido ---' as '';
SELECT pr.id, pr.snomed_id 
FROM Prescripciones pr
LEFT JOIN SnomedCT s ON pr.snomed_id = s.id
WHERE pr.snomed_id IS NOT NULL AND s.id IS NULL;
